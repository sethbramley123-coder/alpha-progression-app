    function addExerciseToSessionDirect(category, name) {

      };

      state.currentSession.push(newEx);
      saveStateToStorage();
      switchTab('workout');
      renderSessionExercises();
    }

    // LIFTING LOG SESSION ENGINE
    function renderSessionExercises() {
      const container = document.getElementById('session-exercises-list');

                  <input type="number" value="${set.reps}" onchange="updateSet(${exIdx}, ${setIdx}, 'reps', this.value)" class="bg-obsidian-900 border border-gray-700 rounded-lg px-2 py-1 text-center font-bold text-white outline-none focus:border-seaGreen-500">
                  <input type="number" value="${set.rpe}" onchange="updateSet(${exIdx}, ${setIdx}, 'rpe', this.value)" class="bg-obsidian-900 border border-gray-700 rounded-lg px-2 py-1 text-center font-bold text-white outline-none focus:border-seaGreen-500">
                  <button onclick="toggleSetDone(${exIdx}, ${setIdx})" class="py-1 rounded-lg font-bold text-xs transition ${set.done ? 'bg-seaGreen-500 text-obsidian-900 shadow-[0_0_8px_rgba(0,210,141,0.5)]' : 'bg-obsidian-800 text-gray-400 border border-gray-700'}">
                    ${set.done ? '<i class="fa-solid fa-check"></i> PR!' : 'Log'}
                  </button>
                </div>
              `).join('')}
            </div>

            <button onclick="addSetToExercise(${exIdx})" class="text-xs text-seaGreen-500 hover:underline font-bold flex items-center gap-1 pt-1">
              <i class="fa-solid fa-plus"></i> Add Set
            </button>
          </div>
        `;
      });

      container.innerHTML = html;
      calculateTotalVolume();

    function addSetToExercise(exIdx) {
      const ex = state.currentSession[exIdx];
      const lastSet = ex.sets[ex.sets.length - 1] || { weight: 60, reps: 10, rpe: 8 };
      ex.sets.push({ ...lastSet, done: false });
      saveStateToStorage();
      renderSessionExercises();
    }

    function updateSet(exIdx, setIdx, field, val) {
      state.currentSession[exIdx].sets[setIdx][field] = parseFloat(val) || 0;
      saveStateToStorage();
      calculateTotalVolume();
    }

    function toggleSetDone(exIdx, setIdx) {
      const set = state.currentSession[exIdx].sets[setIdx];
      set.done = !set.done;
      
      if (set.done) {
        triggerPRCheck(state.currentSession[exIdx].name, set.weight);
        startTimer();
      }

      saveStateToStorage();
      renderSessionExercises();
    }

    function removeExercise(exIdx) {
      state.currentSession.splice(exIdx, 1);
      saveStateToStorage();
      renderSessionExercises();
    }

    function calculateTotalVolume() {
      let totalVol = 0;
      state.currentSession.forEach(ex => {
        ex.sets.forEach(set => {
          if (set.done) totalVol += (set.weight * set.reps);
        });
      });

      const volTag = document.getElementById('total-volume-tag');
      if (volTag) volTag.textContent = `Vol: ${totalVol.toLocaleString()} kg`;
    }

    function triggerPRCheck(exName, weight) {
      const prList = document.getElementById('pr-history-list');
      if (!prList) return;

      const newBadge = `
        <div class="flex items-center justify-between p-2 rounded-lg bg-seaGreen-500/10 border border-seaGreen-500/30">
          <span class="font-bold text-white">${exName}</span>
          <span class="text-seaGreen-500 font-black">${weight} KG PR!</span>
        </div>
      `;

      if (prList.querySelector('p')) prList.innerHTML = "";
      prList.innerHTML = newBadge + prList.innerHTML;
    }

    // TIMER ENGINE
    function formatTime(secs) {
      const m = Math.floor(secs / 60).toString().padStart(2, '0');
      const s = (secs % 60).toString().padStart(2, '0');
      return `${m}:${s}`;
    }

    function updateTimerDisplay() {
      const display = document.getElementById('timer-display');
      if (display) display.textContent = formatTime(state.timer.seconds);
    }

    function toggleTimer() {
      if (state.timer.isRunning) {
        pauseTimer();
      } else {
        startTimer();
      }
    }

    function startTimer() {
      if (state.timer.isRunning) clearInterval(state.timer.interval);
      state.timer.isRunning = true;
      document.getElementById('timer-start-btn').textContent = "Pause Rest";

      state.timer.interval = setInterval(() => {
        if (state.timer.seconds > 0) {
          state.timer.seconds--;
          updateTimerDisplay();
        } else {
          pauseTimer();
          playTimerAlert();
        }
      }, 1000);
    }

    function pauseTimer() {
      clearInterval(state.timer.interval);
      state.timer.isRunning = false;
      const btn = document.getElementById('timer-start-btn');
      if (btn) btn.textContent = "Start Rest";
    }

    function resetTimer() {
      pauseTimer();
      state.timer.seconds = state.timer.initialSeconds;
      updateTimerDisplay();
    }

    function adjustTimer(secs) {
      state.timer.seconds = Math.max(0, state.timer.seconds + secs);
      updateTimerDisplay();
    }

    function setTimerPreset(secs) {
      pauseTimer();
      state.timer.initialSeconds = secs;
      state.timer.seconds = secs;
      updateTimerDisplay();
    }

    function playTimerAlert() {
      try {
        const ctx = new (window.AudioContext || window.webkitAudioContext)();
        const osc = ctx.createOscillator();
        const gain = ctx.createGain();
        osc.connect(gain);
        gain.connect(ctx.destination);
        osc.type = 'sine';
        osc.frequency.value = 880;
        gain.gain.setValueAtTime(0.3, ctx.currentTime);
        osc.start();
        osc.stop(ctx.currentTime + 0.5);
      } catch(e) {
        console.log("Audio block", e);
      }
    }

    // MACRO & STEP CALCULATOR ENGINE
    function renderMacrosAndSteps() {
      let totalP = 0, totalC = 0, totalF = 0;
      state.loggedMeals.forEach(m => {
        totalP += m.protein;
        totalC += m.carbs;
        totalF += m.fats;
      });

      const totalCals = (totalP * 4) + (totalC * 4) + (totalF * 9);

      document.getElementById('cal-burned-display').textContent = `${totalCals.toLocaleString()} / 2,800 kcal`;
      document.getElementById('protein-display').textContent = `${totalP} / 190 g`;
      document.getElementById('carbs-display').textContent = `${totalC} / 270 g`;
      document.getElementById('fats-display').textContent = `${totalF} / 70 g`;

      document.getElementById('protein-progress').style.width = `${Math.min(100, (totalP / 190) * 100)}%`;
      document.getElementById('carbs-progress').style.width = `${Math.min(100, (totalC / 270) * 100)}%`;
      document.getElementById('fats-progress').style.width = `${Math.min(100, (totalF / 70) * 100)}%`;

      renderLoggedMealsList();
      updateStepRing();
    }

    function logMeal() {
      const name = document.getElementById('meal-name-input').value.trim();
      const p = parseInt(document.getElementById('meal-protein-input').value) || 0;
      const c = parseInt(document.getElementById('meal-carbs-input').value) || 0;
      const f = parseInt(document.getElementById('meal-fats-input').value) || 0;

      if (!name) return;

      state.loggedMeals.push({ id: Date.now(), name, protein: p, carbs: c, fats: f });
      saveStateToStorage();
      renderMacrosAndSteps();

      document.getElementById('meal-name-input').value = "";
      document.getElementById('meal-protein-input').value = "";
      document.getElementById('meal-carbs-input').value = "";
      document.getElementById('meal-fats-input').value = "";
    }

    function renderLoggedMealsList() {
      const container = document.getElementById('logged-meals-list');
      if (!container) return;

      if (state.loggedMeals.length === 0) {
        container.innerHTML = `<p class="text-xs text-gray-500 italic">No meals logged today.</p>`;
        return;
      }

      container.innerHTML = state.loggedMeals.map((m, idx) => `
        <div class="flex justify-between items-center p-2 rounded-xl bg-obsidian-900 border border-gray-800 text-xs">
          <div>
            <span class="font-bold text-white block">${m.name}</span>
            <span class="text-gray-400 text-[10px]">P: ${m.protein}g | C: ${m.carbs}g | F: ${m.fats}g</span>
          </div>
          <button onclick="deleteMeal(${idx})" class="text-gray-500 hover:text-rose-400"><i class="fa-solid fa-trash"></i></button>
        </div>
      `).join('');
    }

    function deleteMeal(idx) {
      state.loggedMeals.splice(idx, 1);
      saveStateToStorage();
      renderMacrosAndSteps();
    }

    function updateStepRing() {
      const steps = state.user ? state.user.steps : 8420;
      const target = 10000;
      const percentage = Math.min(1, steps / target);

      const offset = 251.2 - (251.2 * percentage);
      const ring = document.getElementById('step-ring');
      if (ring) ring.style.strokeDashoffset = offset;

      document.getElementById('steps-count').textContent = steps.toLocaleString();
      document.getElementById('distance-display').textContent = `${(steps * 0.00075).toFixed(2)} km`;
      document.getElementById('step-cal-display').textContent = `${Math.round(steps * 0.045)} kcal`;
    }

    function simulateSteps(add) {
      if (!state.user) return;
      state.user.steps += add;
      saveStateToStorage();
      updateStepRing();
    }
